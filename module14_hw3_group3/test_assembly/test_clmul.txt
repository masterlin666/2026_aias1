# test_clmul.txt -- Zbc CLMUL tests

    li x1, 0xB
    li x2, 0xD
    clmul x3, x1, x2        # Expected: x3 = 127 (0x7f)

    li x4, 0xFFFFFFFF
    li x5, 2
    clmul x6, x4, x5        # Expected: x6 = -2 (0xfffffffe)

    li x7, 0x12345678
    li x8, 0
    clmul x9, x7, x8        # Expected: x9 = 0 (0x0)
