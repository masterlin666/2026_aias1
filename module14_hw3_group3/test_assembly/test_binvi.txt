# Test 1: Basic immediate invert
li x1, 0xFFFFFFFF
binvi x2, x1, 0    # Flip bit 0
# Expected: x2 = -2 (0xFFFFFFFE)

# Test 2: High bit immediate
li x1, 0
binvi x3, x1, 15   # Flip bit 15
# Expected: x3 = 32768 (0x00008000)

# Test 3: Toggle multiple times (manual sequence)
li x1, 0
binvi x4, x1, 5
binvi x4, x4, 5    # Toggle back to 0
# Expected: x4 = 0
